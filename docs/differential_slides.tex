\begin{frame}{The corrected null follows constant local path usage}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\vspace{0.1cm}
\begin{block}{Isoform-logit predictor}
\scriptsize A label-independent pooled fit defines the path-ILR Jacobian $J_b\in\mathbb R^{d_b\times(T_g-1)}$. The tested basis is a right inverse, $J_bH_b=I_{d_b}$, and the nuisance basis spans its kernel, $J_bN_b=0$.
\end{block}
\vspace{0.15cm}
\scriptsize The null predictor is $\mathcal F_{bi}^{(0)}=B_0'Z_{bi}'+\sum_jW_{bij}N_b\bm\alpha_{bj}$, while the alternative adds $\sum_jW_{bij}H_b\bm\gamma_{bj}$.
\vspace{0.2cm}
\begin{itemize}\scriptsize
\item $Z_b\in\mathbb R^{I_b\times q_0}$ carries nuisance covariates
\item $W_b\in\mathbb R^{I_b\times q_1}$ carries the tested cell-type or condition design
\item The alternative adds $q_1d_b$ local path coefficients and remains unrestricted; for a pairwise cassette test, $d_b=1$
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\begin{tikzpicture}[x=1cm,y=1cm,>=Latex]
\node[rounded corners=3pt,fill=tlLight,draw=tlGray!55,minimum width=4.5cm,minimum height=1.15cm,align=center] (null) at (0,3.6) {\textbf{Null}\\tangent directions $N_b=\ker(J_b)$};
\node[rounded corners=3pt,fill=tlLightTeal,draw=tlTeal!70,minimum width=4.5cm,minimum height=1.15cm,align=center] (alt) at (0,1.75) {\textbf{Alternative}\\$N_b$ plus right inverse $H_b$};
\draw[->,tlTeal,line width=1.5pt] (null)--node[right,font=\scriptsize,align=left] {add $q_1d_b$\\coefficients} (alt);
\node[rounded corners=3pt,fill=tlNavy,text=white,minimum width=4.5cm,minimum height=.9cm,align=center] at (0,.15) {fit nested EC GLMMs\\calibrate objective gain};
\end{tikzpicture}
\vspace{0.1cm}
\begin{alertblock}{Calibration matters}
\scriptsize The ordinary multilevel chi-square LRT is anti-conservative under structured nulls. The primary mouse result uses simulated nulls, cross-fitted tail modeling, leave-block-out calibration, and BH.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Corrected full-data calibration yields no BH discoveries}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\centering
\metric{tlTeal}{522}{nominal cell-type tests}
\hspace{0.15cm}
\metric{tlOrange}{922}{nominal condition tests}
\vspace{0.55cm}
\metric{tlNavy}{0}{cell-type BH calls}
\hspace{0.15cm}
\metric{tlGold}{0}{condition BH calls}
\end{column}
\begin{column}{0.48\textwidth}
\vspace{0.05cm}
\begin{itemize}\small
\item 4,825 cell-type path partitions and 14,833 condition-within-cell-type tests
\item Aggregate null rejection matches 0.05, 0.01, and 0.001 after cross-fitted calibration
\item Minimum calibrated p-values are 0.00216 for cell type and 0.000148 for condition, above their first BH thresholds
\item The primary logistic-normal likelihood allows pseudobulk extra-multinomial variation shared across the two primer halves
\end{itemize}
\vspace{0.05cm}
\begin{block}{Interpretation}
\scriptsize The corrected tangent null removes the former-basis 238-call set. The earlier structural and literature catalog is hypothesis-generating rather than a validated discovery list.
\end{block}
\end{column}
\end{columns}
\end{frame}

\bottomrule

\vspace{0.4cm}
\begin{block}{What the paired local-path test changes}
\scriptsize Subject-level path ILRs are estimated directly from both primer-specific EC vectors, stabilized by add-two smoothing, variance-moderated across scalar tests, and empirically calibrated with 32 sign-flip families. ``Rep.'' applies BH to the conjunction p-value on each pairwise shared gene universe; $\rho$ is fold correlation of gene--pair $-\log_{10}p$.
\end{block}
\begin{alertblock}{Interpretation}
\scriptsize Every aggregated replicated-gene null family has zero BH calls. Add-two smoothing was selected on these splits, so independent-data confirmation remains necessary.
\end{alertblock}
\end{frame}
